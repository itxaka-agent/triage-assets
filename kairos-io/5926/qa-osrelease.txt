package utils_test

import (
	"errors"
	"os"
	"path/filepath"
	"testing"

	"github.com/kairos-io/kairos/v4/sdk/utils"
)

func TestQAExplicitFileDoesNotLeakHost(t *testing.T) {
	target := filepath.Join(t.TempDir(), "target")
	os.WriteFile(target, []byte("KAIROS_MODEL=generic\nID=ubuntu\n"), 0o644)
	v, err := utils.OSRelease("FLAVOR", target)
	t.Logf("OSRelease(FLAVOR, target) = %q, err=%v", v, err)
	var nf utils.KeyNotFoundErr
	if v != "" || !errors.As(err, &nf) {
		t.Fatalf("want empty + KeyNotFoundErr, got %q, %v", v, err)
	}
}

func TestQANoArgFallsBackToOSRelease(t *testing.T) {
	v, err := utils.OSRelease("FLAVOR")
	t.Logf("OSRelease(FLAVOR) = %q, err=%v", v, err)
	if err != nil || v != "hostflavor" {
		t.Fatalf("want hostflavor from /etc/os-release, got %q, %v", v, err)
	}
}
